% Styles

\tikzset{
  % First stack
  barOutline/.style={
    draw=black,
    line width=0.5pt,
    rounded corners=0pt,
  },
  barBackground/.style={
    fill=gray!15,
    rounded corners=0pt,
  },
  barFill/.style={
    fill=TealBlue,
    rounded corners=0pt,
  },
  rangeLine/.style={
    draw=black!60,
    dotted,
    line width=0.5pt,
  },
  tickLine/.style={
    draw=black,
    line width=0.5pt,
  },
  % Second stack
  barOutlineW/.style={
    draw=black,
    line width=0.5pt,
    rounded corners=0pt,
  },
  barBackgroundW/.style={
    fill=gray!10,
    rounded corners=0pt,
  },
  barFillW/.style={
    fill=TealBlue!50!black,
    rounded corners=0pt,
  },
  % Final eventfulness bar
  eventBarOutline/.style={
    draw=black,
    line width=0.5pt,
    rounded corners=0pt,
  },
  eventBarBackground/.style={
    fill=gray!15,
    rounded corners=0pt,
  },
  eventBarFill/.style={
    fill=Black,
    rounded corners=0pt,
  },
  % Summation fan-in + MediaPipe fan-out
  sumLine/.style={
    draw=black,
    line width=0.6pt,
  },
  sumCircle/.style={
    draw=black,
    line width=0.6pt,
  },
}

\begin{tikzpicture}[x=1cm,y=1cm,font=\small]

  % Shared vertical layout
  \pgfmathtruncatemacro{\TotalRows}{\NumFront + 1 + \NumBack}
  \pgfmathsetmacro{\RowStep}{\BarHeight + \BarGap}
  \pgfmathsetmacro{\yBottom}{-(\TotalRows-1)*\RowStep}
  \pgfmathsetmacro{\MidY}{0.5*(\yBottom + \BarHeight)}

  % MediaPipe circle center coordinates (left side stage)
  \pgfmathsetmacro{\MPX}{- \MPCenterGap}
  \pgfmathsetmacro{\MPY}{\MidY}

  % Frame position (to the left of MediaPipe node)
  \pgfmathsetmacro{\FrameRightX}{\MPX - \FrameToMPCenterGap}
  \pgfmathsetmacro{\FrameX}{\FrameRightX - \FrameWidth}
  \pgfmathsetmacro{\FrameY}{\MPY - 0.5*\FrameHeight}

  % x where MediaPipe connectors end (left of label)
  \pgfmathsetmacro{\MPConnEndX}{\LabelX - \MPConnectorDist}

  %----- Draw frame image -----
  \node[anchor=south west, inner sep=0pt] at (\FrameX,\FrameY)
    {\includegraphics[width=\FrameWidth cm,height=\FrameHeight cm]{\FrameImagePath}};

  %----- Arrow from frame to MediaPipe circle -----
  \draw[->,sumLine] (\FrameRightX,\MPY) -- (\MPX-\MPRadius-0.2,\MPY);

  %----- MediaPipe circle with logo -----
  \draw[sumCircle] (\MPX,\MPY) circle (\MPRadius);
  \node[inner sep=0pt] at (\MPX,\MPY)
    {\includegraphics[width=\MPLogoSize cm]{\MediaPipeLogoPath}};

  %================================================
  % FIRST STACK: B_i \in [0,1]
  %================================================
  \newcommand{\DrawBlendBar}[2]{%
    \pgfmathsetmacro{\y}{-(#1-1)*\RowStep}
    \pgfmathsetmacro{\yMid}{\y + 0.5*\BarHeight}
    \pgfmathsetmacro{\BlendVal}{\BlendValOf{#2}}
    \pgfmathsetmacro{\FilledWidth}{\BlendVal * \BarWidth}

    \fill[barBackground] (0,\y) rectangle (\BarWidth,\y+\BarHeight);
    \draw[barOutline]   (0,\y) rectangle (\BarWidth,\y+\BarHeight);
    \fill[barFill]      (0,\y) rectangle (\FilledWidth,\y+\BarHeight);

    % Label at configurable x
    \node[anchor=east] at (\LabelX,\yMid) {$\BaseName_{#2}$};

    % MediaPipe fan-out connector:
    % same S-style structure as summation connectors, but from MP node to a
    % common x-position left of the labels.
    \draw[sumLine]
      (\MPX+\MPRadius,\MPY)
      .. controls (\MPX+2,\MPY) and (\MPConnEndX-1,\yMid)
      .. (\MPConnEndX,\yMid);
  }

  % Top bars
  \foreach \i [count=\row] in {1,...,\NumFront} {
    \DrawBlendBar{\row}{\i}
  }

  % Ellipsis
  \pgfmathtruncatemacro{\EllipsisRow}{\NumFront + 1}
  \pgfmathsetmacro{\yEll}{-(\EllipsisRow-1)*\RowStep}
  \node at (0.5*\BarWidth,\yEll+0.7*\BarHeight) {$\vdots$};

  % Bottom bars
  \pgfmathtruncatemacro{\BackStartRow}{\NumFront + 2}
  \foreach \k in {1,...,\NumBack} {
    \pgfmathtruncatemacro{\row}{\BackStartRow + \k - 1}
    \pgfmathtruncatemacro{\idx}{\NumBars - \NumBack + \k}
    \DrawBlendBar{\row}{\idx}
  }

  % 0–1 guides for first stack
  \draw[rangeLine] (0,\BarHeight) -- (0,\yBottom);
  \draw[rangeLine] (\BarWidth,\BarHeight) -- (\BarWidth,\yBottom);

  \pgfmathsetmacro{\yTickBase}{\yBottom - 0.1}
  \pgfmathsetmacro{\yTickTop}{\yBottom}
  \draw[tickLine] (0,\yTickBase) -- (0,\yTickTop);
  \draw[tickLine] (\BarWidth,\yTickBase) -- (\BarWidth,\yTickTop);

  \pgfmathsetmacro{\yLabel}{\yTickBase - 0.05}
  \node[anchor=north] at (0,\yLabel) {0};
  \node[anchor=north] at (\BarWidth,\yLabel) {1};

  \pgfmathsetmacro{\BottomY}{\yLabel - 0.7}
  \node[anchor=north] at (0.33*\BarWidth,\BottomY)
    {$\BaseName_i \in [0,1],\quad i=1,\dots,\NumBars$};

  %================================================
  % SECOND STACK: w_i B_i, range [0, w_i]
  %================================================
  % Layout for connectors and summation node
  \pgfmathsetmacro{\ConnX}{\SecondX + \MaxWeight*\BarWidth + \ConnectorGap}
  \pgfmathsetmacro{\SumX}{\ConnX + \SumGap}
  \pgfmathsetmacro{\SumY}{0.5*(\yBottom + \BarHeight)}
  \pgfmathsetmacro{\SumXin}{\SumX - \SumRadius}

  \newcommand{\DrawWeightedBar}[2]{%
    \pgfmathsetmacro{\y}{-(#1-1)*\RowStep}
    \pgfmathsetmacro{\BlendVal}{\BlendValOf{#2}}
    \pgfmathsetmacro{\WeightVal}{\WeightValOf{#2}}

    \pgfmathsetmacro{\FullWidth}{\WeightVal * \BarWidth}   % [0,w_i]
    \pgfmathsetmacro{\FilledWidth}{\BlendVal * \FullWidth} % w_i B_i
    \pgfmathsetmacro{\xRight}{\SecondX + \FullWidth}
    \pgfmathsetmacro{\yMid}{\y + 0.5*\BarHeight}

    % Background and outline
    \fill[barBackgroundW] (\SecondX,\y)
       rectangle ++(\FullWidth,\BarHeight);
    \draw[barOutlineW]    (\SecondX,\y)
       rectangle ++(\FullWidth,\BarHeight);
    % Filled portion
    \fill[barFillW]       (\SecondX,\y)
       rectangle ++(\FilledWidth,\BarHeight);

    % Left label: w_i B_i
    \node[anchor=east] at (\SecondX-0.2,\yMid)
      {$\WeightName_{#2}\,\BaseName_{#2}$};

    % Horizontal tick + label at bar right end (w_i)
    \draw[tickLine] (\xRight,\yMid) -- (\xRight+0.12,\yMid);
    \node[anchor=west] at (\xRight+0.14,\yMid)
      {$\WeightName_{#2}$};

    % Connector: from common X (same for all) to summation node
    \draw[sumLine]
      (\ConnX,\yMid)
      .. controls (\ConnX+1,\yMid) and (\SumX-1.5,\SumY)
      .. (\SumXin,\SumY);
  }

  % Top weighted bars
  \foreach \i [count=\row] in {1,...,\NumFront} {
    \DrawWeightedBar{\row}{\i}
  }

  % Ellipsis in second stack
  \pgfmathsetmacro{\yEllW}{-(\EllipsisRow-1)*\RowStep}
  \node at (\SecondX+0.5*\BarWidth,\yEllW+0.7*\BarHeight) {$\vdots$};

  % Bottom weighted bars
  \foreach \k in {1,...,\NumBack} {
    \pgfmathtruncatemacro{\row}{\BackStartRow + \k - 1}
    \pgfmathtruncatemacro{\idx}{\NumBars - \NumBack + \k}
    \DrawWeightedBar{\row}{\idx}
  }

  % Vertical zero line for second stack
  \draw[rangeLine] (\SecondX,\BarHeight) -- (\SecondX,\yBottom);

  % Vertical tick + "0" under bottom-most bar for second stack
  \pgfmathsetmacro{\yTickBaseW}{\yBottom - 0.1}
  \pgfmathsetmacro{\yTickTopW}{\yBottom}
  \draw[tickLine] (\SecondX,\yTickBaseW) -- (\SecondX,\yTickTopW);
  \pgfmathsetmacro{\yLabelW}{\yTickBaseW - 0.05}
  \node[anchor=north] at (\SecondX,\yLabelW) {0};

  % Bottom description for w_i
  \pgfmathsetmacro{\BottomYw}{\BottomY}
  \node[anchor=north] at (\SecondX+0.25*\BarWidth,\BottomYw)
    {$\WeightName_i \in \mathbb{R},\quad i=1,\dots,\NumBars$};

  %================================================
  % Summation node
  %================================================
  \draw[sumCircle] (\SumX,\SumY) circle (\SumRadius);
  \node at (\SumX,\SumY) {$\sum$};

  %================================================
  % Final eventfulness bar: sum_i w_i B_i in [0, W]
  %================================================
  \pgfmathsetmacro{\EventX}{\SumX + \SumRadius + \EventGap}
  \pgfmathsetmacro{\EventY}{\SumY - 0.5*\BarHeight}
  \pgfmathsetmacro{\EventFillWidth}{(\EventVal/\TotalWeight) * \EventBarWidth}

  % Background & outline
  \fill[eventBarBackground] (\EventX,\EventY)
    rectangle ++(\EventBarWidth,\BarHeight);
  \draw[eventBarOutline] (\EventX,\EventY)
    rectangle ++(\EventBarWidth,\BarHeight);
  % Filled portion
  \fill[eventBarFill] (\EventX,\EventY)
    rectangle ++(\EventFillWidth,\BarHeight);

  % Left label: sum_i w_i B_i
  \node[anchor=east] at (\EventX-0.2,\EventY+0.5*\BarHeight)
    {$\displaystyle\sum_{i=1}^{\NumBars} \WeightName_i \BaseName_i$};

  % Ticks and labels under final bar
  \pgfmathsetmacro{\EventTickBase}{\EventY - 0.10}
  \pgfmathsetmacro{\EventTickTop}{\EventY}
  \pgfmathsetmacro{\EventRightX}{\EventX + \EventBarWidth}
  \draw[tickLine] (\EventX,\EventTickBase) -- (\EventX,\EventTickTop);
  \draw[tickLine] (\EventRightX,\EventTickBase) -- (\EventRightX,\EventTickTop);

  \pgfmathsetmacro{\EventLabelY}{\EventTickBase - 0.05}
  \node[anchor=north] at (\EventX,\EventLabelY) {0};
  \node[anchor=north] at (\EventRightX,\EventLabelY) {$W$};

  % Explanation for W under the final bar
  \pgfmathsetmacro{\EventExplainY}{\EventLabelY - 0.45}
  \node[anchor=north] at (\EventX+0.5*\EventBarWidth,\EventExplainY)
    {$W = \displaystyle\sum_{i=1}^{\NumBars} \WeightName_i$};

\end{tikzpicture}
