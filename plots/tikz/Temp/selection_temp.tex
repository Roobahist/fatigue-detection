\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{calc,arrows.meta}

\begin{document}

% --------------------------------------------------
% Configuration
% --------------------------------------------------
\def\NumWindows{8}     % number of windows in each column
\def\TopK{3}           % how many to select in top-K methods

% Geometry
\def\ColumnSpacing{6}  % horizontal spacing between the three methods
\def\BarWidth{0.6}     % width of each window/bar
\def\BarSpacing{0.9}   % center-to-center spacing of windows in a column
\def\YScale{0.4}       % scale factor for bar heights

% Colors
\definecolor{basegray}{gray}{0.7}
\def\HighlightColorAvg{cyan}
\def\HighlightColorMax{magenta}
\def\HighlightColorRand{orange}

\begin{tikzpicture}[x=0.5cm,y=0.5cm]

% ==================================================
% Helper: draw one column (method) given:
%  - x offset
%  - label text
%  - a scheme for heights
%  - a scheme for which bars are highlighted
% ==================================================

% -------- Column 1: sort by avg, pick top-K --------
\def\xOffsetA{0}

% Title
\node[font=\scriptsize, anchor=south]
  at (\xOffsetA + \NumWindows*\BarSpacing/2, 7)
  {Method 1: sort by $\mathrm{avg}_k$ and pick top $K$};

% y-axis label (conceptual)
\node[font=\scriptsize, anchor=east, rotate=90]
  at (\xOffsetA - 1.0, 3) {$\mathrm{avg}_k$ (sorted)};  

\foreach \k in {1,...,\NumWindows} {%
  % Rank index r = 1..NumWindows (left to right)
  \pgfmathsetmacro{\r}{\k}
  % avg score: simple decreasing sequence so monotone is obvious
  \pgfmathsetmacro{\avg}{\NumWindows+1-\r} % 8,7,6,...,1
  \pgfmathsetmacro{\h}{\avg*\YScale}

  % x-position of bar
  \pgfmathsetmacro{\xL}{\xOffsetA + (\r-1)*\BarSpacing}
  \pgfmathsetmacro{\xR}{\xL + \BarWidth}

  % Decide fill for top-K (bars with smallest r, highest avg)
  \ifnum\r>\TopK
    \def\BarDraw{basegray}
    \def\BarFill{basegray}
    \def\BarFillOpacity{0.4}
  \else
    \def\BarDraw{\HighlightColorAvg}
    \def\BarFill{\HighlightColorAvg}
    \def\BarFillOpacity{0.9}
  \fi

  % Draw bar as a "window"
  \draw[draw=\BarDraw, fill=\BarFill, fill opacity=\BarFillOpacity]
    (\xL,0) rectangle (\xR,\h);

  % (Optional) tiny tick for rank index under bars
  \node[font=\tiny, anchor=north]
    at (0.5*(\xL+\xR), -0.3) {\r};
}

% Bracket / label for top-K region
\draw[\HighlightColorAvg, line width=0.7pt]
  (\xOffsetA - 0.5, 5.5) -- (\xOffsetA - 0.5, 6.5);
\draw[\HighlightColorAvg, line width=0.7pt]
  (\xOffsetA - 0.5, 6.5) -- (\xOffsetA + \TopK*\BarSpacing - 0.4, 6.5);
\draw[\HighlightColorAvg, line width=0.7pt]
  (\xOffsetA + \TopK*\BarSpacing - 0.4, 5.5) -- (\xOffsetA + \TopK*\BarSpacing - 0.4, 6.5);
\node[font=\scriptsize, anchor=south, text=\HighlightColorAvg]
  at (\xOffsetA + \TopK*\BarSpacing/2 - 0.2, 6.6)
  {$\text{select top }K$};

% ==================================================
% Column 2: sort by max, pick top-K
% ==================================================

\pgfmathsetmacro{\xOffsetB}{\ColumnSpacing}

\node[font=\scriptsize, anchor=south]
  at (\xOffsetB + \NumWindows*\BarSpacing/2, 7)
  {Method 2: sort by $\mathrm{max}_k$ and pick top $K$};

\node[font=\scriptsize, anchor=east, rotate=90]
  at (\xOffsetB - 1.0, 3) {$\mathrm{max}_k$ (sorted)};  

\foreach \k in {1,...,\NumWindows} {%
  % Here we fake a different sorted order by using a different pattern
  % e.g. slightly flatter distribution so it looks distinct.
  \pgfmathsetmacro{\r}{\k}
  \pgfmathsetmacro{\maxv}{1.5*(\NumWindows+1-\r) + 2} % just a different shape
  \pgfmathsetmacro{\h}{\maxv*\YScale*0.6}             % scale down a bit

  \pgfmathsetmacro{\xL}{\xOffsetB + (\r-1)*\BarSpacing}
  \pgfmathsetmacro{\xR}{\xL + \BarWidth}

  \ifnum\r>\TopK
    \def\BarDraw{basegray}
    \def\BarFill{basegray}
    \def\BarFillOpacity{0.4}
  \else
    \def\BarDraw{\HighlightColorMax}
    \def\BarFill{\HighlightColorMax}
    \def\BarFillOpacity{0.9}
  \fi

  \draw[draw=\BarDraw, fill=\BarFill, fill opacity=\BarFillOpacity]
    (\xL,0) rectangle (\xR,\h);

  \node[font=\tiny, anchor=north]
    at (0.5*(\xL+\xR), -0.3) {\r};
}

\draw[\HighlightColorMax, line width=0.7pt]
  (\xOffsetB - 0.5, 5.2) -- (\xOffsetB - 0.5, 6.2);
\draw[\HighlightColorMax, line width=0.7pt]
  (\xOffsetB - 0.5, 6.2) -- (\xOffsetB + \TopK*\BarSpacing - 0.4, 6.2);
\draw[\HighlightColorMax, line width=0.7pt]
  (\xOffsetB + \TopK*\BarSpacing - 0.4, 5.2) -- (\xOffsetB + \TopK*\BarSpacing - 0.4, 6.2);
\node[font=\scriptsize, anchor=south, text=\HighlightColorMax]
  at (\xOffsetB + \TopK*\BarSpacing/2 - 0.2, 6.3)
  {$\text{select top }K$};

% ==================================================
% Column 3: sorted by avg, but random selection p_k ∝ avg_k
% ==================================================

\pgfmathsetmacro{\xOffsetC}{2*\ColumnSpacing}

\node[font=\scriptsize, anchor=south]
  at (\xOffsetC + \NumWindows*\BarSpacing/2, 7)
  {Method 3: sample $K$ windows with $p_k \propto \mathrm{avg}_k$};

\node[font=\scriptsize, anchor=east, rotate=90]
  at (\xOffsetC - 1.0, 3) {$\mathrm{avg}_k$ (sorted)};  

% Define which ranks are randomly selected (non-contiguous)
% Example: select ranks 1,3,4,6 for K=4 just to show scattering
\def\RandSelA{1}
\def\RandSelB{3}
\def\RandSelC{4}
\def\RandSelD{6}

\foreach \k in {1,...,\NumWindows} {%
  \pgfmathsetmacro{\r}{\k}
  % Reuse same avg pattern as Column 1 so order is identical
  \pgfmathsetmacro{\avg}{\NumWindows+1-\r}
  \pgfmathsetmacro{\h}{\avg*\YScale}

  \pgfmathsetmacro{\xL}{\xOffsetC + (\r-1)*\BarSpacing}
  \pgfmathsetmacro{\xR}{\xL + \BarWidth}

  % Check if this rank is "sampled"
  \def\BarDraw{basegray}
  \def\BarFill{basegray}
  \def\BarFillOpacity{0.3}

  \ifnum\r=\RandSelA
    \def\BarDraw{\HighlightColorRand}
    \def\BarFill{\HighlightColorRand}
    \def\BarFillOpacity{0.9}
  \fi
  \ifnum\r=\RandSelB
    \def\BarDraw{\HighlightColorRand}
    \def\BarFill{\HighlightColorRand}
    \def\BarFillOpacity{0.9}
  \fi
  \ifnum\r=\RandSelC
    \def\BarDraw{\HighlightColorRand}
    \def\BarFill{\HighlightColorRand}
    \def\BarFillOpacity{0.9}
  \fi
  \ifnum\r=\RandSelD
    \def\BarDraw{\HighlightColorRand}
    \def\BarFill{\HighlightColorRand}
    \def\BarFillOpacity{0.9}
  \fi

  \draw[draw=\BarDraw, fill=\BarFill, fill opacity=\BarFillOpacity]
    (\xL,0) rectangle (\xR,\h);

  \node[font=\tiny, anchor=north]
    at (0.5*(\xL+\xR), -0.3) {\r};
}

% Legend-ish explanation next to Column 3
\node[font=\scriptsize, align=left, anchor=north west]
  at (\xOffsetC + \NumWindows*\BarSpacing + 1, 5.5)
  {$p_k = \dfrac{\mathrm{avg}_k}{\sum_j \mathrm{avg}_j}$\\
   highlighted bars: sampled windows};

\end{tikzpicture}

\end{document}
